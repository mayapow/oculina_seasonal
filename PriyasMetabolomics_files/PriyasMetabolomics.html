num_df <- oculina[, -c(1:27, 31358:31386)]
metapheno <- oculina$bleaching
meta_id <- oculina$id


braycurtis <- vegdist(num_df, method = "bray")
pcoa_result <- cmdscale(braycurtis, eig = TRUE)
eig_vals <- pcoa_result$eig
variance_explained <- eig_vals/sum(eig_vals) * 100

pcoa_df_ocu <- data.frame(
  PC1 = pcoa_result$points[, 1],
  PC2 = pcoa_result$points[, 2],
  Phenotype = metapheno,
  ID = meta_id
)
