#generate IDs for new data for duplicates
id_map <- tribble(
  ~old_id, ~new_id,
  "S1-A",  "SX1-A",
  "S1-S",  "SX1-S",
  "S15-A", "SX2-A",
  "S15-S", "SX2-S",
  "A15-A", "AX4-A",
  "A15-S", "AX4-S"
)

#create duplicate dataframe to add the phys data to these IDs
physio_duplicates <- physio %>%
  inner_join(id_map, by = c("sample_id" = "old_id")) %>%
  mutate(sample_id = new_id) %>% #rename to sample id to merge
  select(-new_id) #remove new id variable

#add rows into physio dataframe to join with metabolomics
physio_metab <- bind_rows(physio, physio_duplicates)
